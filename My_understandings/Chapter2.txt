Four things to master:

Method — what you want to do (GET/POST/PUT/PATCH/DELETE/HEAD/OPTIONS)

URL — which resource (/api/tasks/42/)

Headers — metadata (auth, content type, accept, cache)

Body — the payload (JSON, form data, files)

function hit {
    param([string]$Url, [string]$Method = "GET", [hashtable]$Headers = @{}, [string]$Body = $null)
    Write-Host "`n=== $Method $Url ===" -ForegroundColor Cyan
    $args = @("-s", "-i", "-X", $Method)
    foreach ($k in $Headers.Keys) { $args += @("-H", "$k`: $($Headers[$k])") }
    if ($Body) { $args += @("-d", $Body) }
    $args += $Url
    & curl.exe @args
    Write-Host ""
}

. $PROFILE
